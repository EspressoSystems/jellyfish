73impl<E, H, T, I, N> DigestAlgorithm<E, I, NamespacedHash<T, N>> for NamespacedHasher<H, E, I, T, N>
74where
75    E: Element + Namespaced<Namespace = N>,
76    I: Index,
77    N: Namespace,
78    T: NodeValue,
79    H: DigestAlgorithm<E, I, T> + BindNamespace<E, I, T, N>,
80{
81    // Assumes that data is sorted by namespace, will be enforced by "append"
82    fn digest(data: &[NamespacedHash<T, N>]) -> Result<NamespacedHash<T, N>, MerkleTreeError> {
83        if data.is_empty() {
84            return Ok(NamespacedHash::default());
85        }
86        let first_node = data[0];
87        let min_namespace = first_node.min_namespace;
88        let mut max_namespace = first_node.max_namespace;
89        let mut nodes = vec![H::generate_namespaced_commitment(first_node)];
90        for node in &data[1..] {
91            if node == &NamespacedHash::default() {
92                continue;
93            }
94            // Ensure that namespaced nodes are sorted
95            if node.min_namespace < max_namespace {
96                return Err(MerkleTreeError::InconsistentStructureError(
97                    "Namespace Merkle tree leaves are out of order".to_string(),
98                ));
99            }
100            max_namespace = node.max_namespace;
101            nodes.push(H::generate_namespaced_commitment(*node));
102        }
103
104        let inner_hash = H::digest(&nodes)?;
105
106        Ok(NamespacedHash::new(
107            min_namespace,
108            max_namespace,
109            inner_hash,
110        ))
111    }
112
113    fn digest_leaf(pos: &I, elem: &E) -> Result<NamespacedHash<T, N>, MerkleTreeError> {
114        let namespace = elem.get_namespace();
115        let hash = H::digest_leaf(pos, elem)?;
116        Ok(NamespacedHash::new(namespace, namespace, hash))
117    }
118}

120impl<E, I, N> BindNamespace<E, I, Sha3Node, N> for Sha3Digest
121where
122    E: Element + CanonicalSerialize,
123    I: Index,
124    N: Namespace,
125{
126    // TODO ensure the hashing of (min,max,hash) is collision resistant
127    fn generate_namespaced_commitment(namespaced_hash: NamespacedHash<Sha3Node, N>) -> Sha3Node {
128        let mut hasher = Sha3_256::new();
129        let mut writer = Vec::new();
130        namespaced_hash
131            .min_namespace
132            .serialize_compressed(&mut writer)
133            .unwrap();
134        namespaced_hash
135            .max_namespace
136            .serialize_compressed(&mut writer)
137            .unwrap();
138        namespaced_hash
139            .hash
140            .serialize_compressed(&mut writer)
141            .unwrap();
142        hasher.update(&mut writer);
143        Sha3Node(hasher.finalize().into())
144    }
