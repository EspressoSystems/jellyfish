29/// Namespaced Merkle Tree where leaves are sorted by a namespace identifier.
30/// The data structure supports namespace inclusion proofs.
31pub trait NamespacedMerkleTreeScheme: AppendableMerkleTreeScheme
32where
33    Self::Element: Namespaced,
34{
35    /// Namespace proof type
36    type NamespaceProof: NamespaceProof;
37    /// Namespace type
38    type NamespaceId: Namespace;
39
40    /// Returns the entire set of leaves corresponding to a given namespace and
41    /// a completeness proof.
42    fn get_namespace_proof(&self, namespace: Self::NamespaceId) -> Self::NamespaceProof;
43
44    /// Verifies the completeness proof for a given set of leaves and a
45    /// namespace.
46    fn verify_namespace_proof(
47        &self,
48        proof: &Self::NamespaceProof,
49        namespace: Self::NamespaceId,
50    ) -> Result<VerificationResult, MerkleTreeError>;
51}

53/// Completeness proof for a namespace
54pub trait NamespaceProof {
55    /// Namespace type
56    type Namespace: Namespace;
57    /// Namespaced leaf
58    type Leaf: Element + Namespaced<Namespace = Self::Namespace>;
59    /// Internal node value
60    type Node: NodeValue;
61
62    /// Return the set of leaves associated with this Namespace proof
63    fn get_namespace_leaves(&self) -> Vec<&Self::Leaf>;
64
65    /// Verify a namespace proof
66    fn verify(
67        &self,
68        root: &NamespacedHash<Self::Node, Self::Namespace>,
69        namespace: Self::Namespace,
70    ) -> Result<VerificationResult, MerkleTreeError>;
71}

122type InnerTree<E, H, T, N, const ARITY: usize> =
123    MerkleTree<E, NamespacedHasher<H, E, u64, T, N>, u64, ARITY, NamespacedHash<T, N>>;
124
125type NamespaceRanges<N> = BTreeMap<N, Range<u64>>;

127#[derive(Debug, Clone, Serialize, Deserialize, Eq, PartialEq, Hash)]
128#[serde(bound = "E: CanonicalSerialize + CanonicalDeserialize,
129                 T: CanonicalSerialize + CanonicalDeserialize")]
130/// NMT
131pub struct NMT<E, H, const ARITY: usize, N, T>
132where
133    H: DigestAlgorithm<E, u64, T> + BindNamespace<E, u64, T, N>,
134    E: Element + Namespaced<Namespace = N>,
135    T: NodeValue,
136    N: Namespace,
137{
138    namespace_ranges: NamespaceRanges<N>,
139    inner: InnerTree<E, H, T, N, ARITY>,
140}

142impl<E, H, const ARITY: usize, N, T> NMT<E, H, ARITY, N, T>
143where
144    H: DigestAlgorithm<E, u64, T> + BindNamespace<E, u64, T, N>,
145    E: Element + Namespaced<Namespace = N>,
146    T: NodeValue,
147    N: Namespace,
148{
149    /// Initializze an empty NMT
150    pub fn new(height: usize) -> Self {
151        let namespace_ranges: BTreeMap<N, Range<u64>> = BTreeMap::new();
152        let inner = InnerTree::<E, H, T, N, ARITY>::new(height);
153        NMT {
154            inner,
155            namespace_ranges,
156        }
157    }
158
159    /// Construct an NMT from elements.
160    pub fn from_elems(
161        height: Option<usize>,
162        elems: impl IntoIterator<Item = impl Borrow<E>>,
163    ) -> Result<Self, MerkleTreeError> {
164        let mut namespace_ranges: BTreeMap<N, Range<u64>> = BTreeMap::new();
165        let leaves =
166            NMT::<E, H, ARITY, N, T>::update_namespace_metadata(&mut namespace_ranges, elems)?;
167        let inner = InnerTree::<E, H, T, N, ARITY>::from_elems(height, leaves)?;
168        Ok(NMT {
169            inner,
170            namespace_ranges,
171        })
172    }
173}

184    type NodeValue = NamespacedHash<T, N>;
185    type MembershipProof = <InnerTree<E, H, T, N, ARITY> as MerkleTreeScheme>::MembershipProof;
186    type BatchMembershipProof =

188    const ARITY: usize = <InnerTree<E, H, T, N, ARITY> as MerkleTreeScheme>::ARITY;

190

193    }
194
195    fn capacity(&self) -> num_bigint::BigUint {

197    }
198

200        self.inner.num_leaves()
201    }
202
203    fn commitment(&self) -> Self::Commitment {
204        self.inner.commitment()
205    }
206
207    fn lookup(
208        &self,
209        pos: impl Borrow<Self::Index>,
210    ) -> super::LookupResult<Self::Element, Self::MembershipProof, ()> {
211        self.inner.lookup(pos)
212    }
213
214    fn verify(
215        commitment: impl Borrow<Self::Commitment>,
216        pos: impl Borrow<Self::Index>,
217        proof: impl Borrow<Self::MembershipProof>,
218    ) -> Result<VerificationResult, MerkleTreeError> {
219        <InnerTree<E, H, T, N, ARITY> as MerkleTreeScheme>::verify(commitment, pos, proof)
220    }
221
222    fn iter(&self) -> MerkleTreeIter<Self::Element, Self::Index, Self::NodeValue> {
223        self.inner.iter()
224    }
225}

227impl<E, H, const ARITY: usize, N, T> AppendableMerkleTreeScheme for NMT<E, H, ARITY, N, T>
228where
229    H: DigestAlgorithm<E, u64, T> + BindNamespace<E, u64, T, N>,
230    E: Element + Namespaced<Namespace = N>,
231    T: NodeValue,
232    N: Namespace,
233{
234    fn extend(
235        &mut self,
236        elems: impl IntoIterator<Item = impl core::borrow::Borrow<Self::Element>>,
237    ) -> Result<(), MerkleTreeError> {
238        let leaves =
239            NMT::<E, H, ARITY, N, T>::update_namespace_metadata(&mut self.namespace_ranges, elems)?;
240        self.inner.extend(leaves)
241    }
242
243    fn push(
244        &mut self,
245        elem: impl core::borrow::Borrow<Self::Element>,
246    ) -> Result<(), MerkleTreeError> {
247        self.extend([elem])
248    }
249}

251impl<E, H, const ARITY: usize, N, T> NMT<E, H, ARITY, N, T>
252where
253    H: DigestAlgorithm<E, u64, T> + BindNamespace<E, u64, T, N>,
254    E: Element + Namespaced<Namespace = N>,
255    T: NodeValue,
256    N: Namespace,
257{
258    // Helper function to lookup a proof that should be in the tree because of
259    // NMT invariants
260    fn lookup_proof(&self, idx: u64) -> MerkleProof<E, u64, NamespacedHash<T, N>, ARITY> {
261        if let LookupResult::Ok(_, proof) = self.inner.lookup(idx) {
262            proof
263        } else {
264            // The NMT is malformed, we cannot recover
265            panic!()
266        }
267    }
268
269    /// Helper function to return an iterator over the leaves in the tree
270    pub fn leaves(&self) -> impl ExactSizeIterator<Item = E> + '_ {
271        (0..self.num_leaves() as usize).map(|idx| match self.inner.lookup(idx as u64) {
272            LookupResult::Ok(elem, _) => elem,
273            _ => panic!("NMT variant violated: every leaf in the tree should be occupied"),
274        })
275    }
276
277    // Helper function to keep namespace metadata in sync with new leaves,
278    // Returns cached leaf references so that the inner merkle tree can append
279    // them
280    fn update_namespace_metadata(
281        namespace_ranges: &mut NamespaceRanges<N>,
282        elems: impl IntoIterator<Item = impl Borrow<E>>,
283    ) -> Result<Vec<impl Borrow<E>>, MerkleTreeError> {
284        let (mut max_namespace, start_idx) = namespace_ranges
285            .iter()
286            .next_back()
287            .map(|(n, range)| (*n, range.end))
288            .unwrap_or((<N as Namespace>::min(), 0));
289        let mut leaves = Vec::new();
290        for (idx, elem) in elems.into_iter().enumerate() {
291            let ns = elem.borrow().get_namespace();
292            let idx = start_idx + idx as u64;
293            if ns < max_namespace {
294                return Err(MerkleTreeError::InconsistentStructureError(
295                    "Namespace leaves must be pushed in sorted order".into(),
296                ));
297            }
298            match namespace_ranges.entry(ns) {
299                Entry::Occupied(entry) => {
300                    entry.into_mut().end = idx + 1;
301                },
302                Entry::Vacant(entry) => {
303                    entry.insert(idx..idx + 1);
304                },
305            }
306            max_namespace = ns;
307            leaves.push(elem);
308        }
309        Ok(leaves)
310    }
311}

313impl<E, H, const ARITY: usize, N, T> NamespacedMerkleTreeScheme for NMT<E, H, ARITY, N, T>
314where
315    H: DigestAlgorithm<E, u64, T> + BindNamespace<E, u64, T, N> + Clone,
316    E: Element + Namespaced<Namespace = N>,
317    T: NodeValue,
318    N: Namespace,
319{
320    type NamespaceId = N;
321    type NamespaceProof = NaiveNamespaceProof<E, T, ARITY, N, H>;
322
323    fn get_namespace_proof(&self, namespace: Self::NamespaceId) -> Self::NamespaceProof {
324        let ns_range = self.namespace_ranges.get(&namespace);
325        let mut proofs = Vec::new();
326        let mut left_boundary_proof = None;
327        let mut right_boundary_proof = None;
328        let proof_type;
329        let mut first_index = None;
330        if let Some(ns_range) = ns_range {
331            proof_type = NamespaceProofType::Presence;
332            for i in ns_range.clone() {
333                if first_index.is_none() {
334                    first_index = Some(i);
335                }
336                proofs.push(self.lookup_proof(i));
337            }
338            let left_index = first_index.unwrap_or(0);
339            let right_index = left_index + proofs.len() as u64;
340            if left_index > 0 {
341                left_boundary_proof = Some(self.lookup_proof(left_index - 1));
342            }
343            if right_index < self.num_leaves() - 1 {
344                right_boundary_proof = Some(self.lookup_proof(right_index + 1));
345            }
346        } else {
347            proof_type = NamespaceProofType::Absence;
348            // If there is a namespace in the tree greater than our target
349            // namespace at some index i, prove that the
350            // target namespace is empty by providing proofs of leaves at index
351            // i and i - 1
352            if let Some((_, range)) = self.namespace_ranges.range(namespace..).next() {
353                let i = range.start;
354                // If i == 0, the target namespace is less than the tree's
355                // minimum namespace
356                if i > 0 {
357                    left_boundary_proof = Some(self.lookup_proof(i - 1));
358                    right_boundary_proof = Some(self.lookup_proof(i));
359                }
360            }
361        }
362        NaiveNamespaceProof {
363            proof_type,
364            proofs,
365            left_boundary_proof,
366            right_boundary_proof,
367            first_index: first_index.unwrap_or(0),
368            phantom: PhantomData,
369        }
370    }
371
372    fn verify_namespace_proof(
373        &self,
374        proof: &Self::NamespaceProof,
375        namespace: Self::NamespaceId,
376    ) -> Result<VerificationResult, MerkleTreeError> {
377        proof.verify(&self.commitment().digest(), namespace)
378    }
379}

423    fn test_namespaced_hash() {
424        let num_leaves = 5;
425        let leaves: Vec<Leaf> = (0..num_leaves).map(Leaf::new).collect();

427        // Ensure that leaves are digested correctly
428        let mut hashes = leaves
429            .iter()
430            .enumerate()
431            .map(|(idx, leaf)| Hasher::digest_leaf(&(idx as u64), leaf))
432            .collect::<Result<Vec<_>, MerkleTreeError>>()
433            .unwrap();
434        assert_eq!((hashes[0].min_namespace, hashes[0].max_namespace), (0, 0));
435
436        // Ensure that sorted internal nodes are digested correctly
437        let hash = Hasher::digest(&hashes).unwrap();
438        assert_eq!(

440            (0, num_leaves - 1)
441        );

443        // Ensure that digest errors when internal nodes are not sorted by
444        // namespace
445        hashes[0] = hashes[hashes.len() - 1];
446        assert!(Hasher::digest(&hashes).is_err());

493    fn test_nmt_with_build_type(build_type: BuildType) {
494        let namespaces = [1, 2, 2, 2, 4, 4, 4, 5];
495        let leaves: Vec<Leaf> = namespaces.iter().map(|i| Leaf::new(*i)).collect();
496        let first_ns = namespaces[0];
497        let last_ns = namespaces[namespaces.len() - 1];
498        let internal_ns = namespaces[1];
499        let tree = build_tree(&leaves, build_type);
500        let left_proof = tree.get_namespace_proof(first_ns);
501        let right_proof = tree.get_namespace_proof(last_ns);
502        let mut internal_proof = tree.get_namespace_proof(internal_ns);
503
504        // Check all of the leaves
505        let fetched_leaves = tree.leaves().collect::<Vec<Leaf>>();
506        assert_eq!(fetched_leaves, leaves);
507
508        // Check namespace proof on the left boundary
509        assert!(tree
510            .verify_namespace_proof(&left_proof, first_ns)
511            .unwrap()
512            .is_ok());
513
514        // Check namespace proof on the right boundary
515        assert!(tree
516            .verify_namespace_proof(&right_proof, last_ns)
517            .unwrap()
518            .is_ok());
519
520        // Check namespace proof for some internal namespace
521        assert!(tree
522            .verify_namespace_proof(&internal_proof, internal_ns)
523            .unwrap()
524            .is_ok());
525
526        // Assert that namespace proof fails for a different namespace
527        assert!(tree
528            .verify_namespace_proof(&left_proof, 2)
529            .unwrap()
530            .is_err());
531
532        // Sanity check that the leaves returned by the proof are correct
533        let internal_leaves: Vec<Leaf> = internal_proof
534            .get_namespace_leaves()
535            .into_iter()
536            .copied()
537            .collect();
538        let raw_leaves_for_ns = &leaves[1..4];
539        assert_eq!(raw_leaves_for_ns, internal_leaves);
540
541        // Check that a namespace proof fails if one of the leaves is removed
542        internal_proof.proofs.remove(1);
543        assert!(tree
544            .verify_namespace_proof(&internal_proof, internal_ns)
545            .unwrap()
546            .is_err());
547
548        // Check the simple absence proof case when the namespace falls outside
549        // of the tree range (namespace > root.max_namespace)
550        let absence_proof = tree.get_namespace_proof(last_ns + 1);
551        let leaves: Vec<Leaf> = absence_proof
552            .get_namespace_leaves()
553            .into_iter()
554            .cloned()
555            .collect();
556        assert!(tree
557            .verify_namespace_proof(&absence_proof, last_ns + 1)
558            .unwrap()
559            .is_ok());
560        assert_eq!(leaves, []);
561
562        // Check the simple absence proof case when the namespace falls outside
563        // of the tree range (namespace < root.min_namespace)
564        let absence_proof = tree.get_namespace_proof(first_ns - 1);
565        let leaves: Vec<Leaf> = absence_proof
566            .get_namespace_leaves()
567            .into_iter()
568            .cloned()
569            .collect();
570        assert!(tree
571            .verify_namespace_proof(&absence_proof, first_ns - 1)
572            .unwrap()
573            .is_ok());
574        assert_eq!(leaves, []);
575
576        // Check absence proof case when the namespace falls inside of the tree
577        // range
578        let absence_proof = tree.get_namespace_proof(3);
579        let leaves: Vec<Leaf> = absence_proof
580            .get_namespace_leaves()
581            .into_iter()
582            .cloned()
583            .collect();
584        assert!(tree
585            .verify_namespace_proof(&absence_proof, 3)
586            .unwrap()
587            .is_ok());
588        assert_eq!(leaves, []);
589
590        // Ensure that the absence proof fails when the boundaries are not
591        // provided
592        let mut malformed_proof = absence_proof.clone();
593        malformed_proof.left_boundary_proof = None;
594        assert!(tree.verify_namespace_proof(&malformed_proof, 3).is_err());
595        let mut malformed_proof = absence_proof.clone();
596        malformed_proof.right_boundary_proof = None;
597        assert!(tree.verify_namespace_proof(&malformed_proof, 3).is_err());
598
599        // Ensure that the absence proof returns a verification error when one
600        // of the boundary proofs is incorrect
601        let mut malicious_proof = absence_proof.clone();
602        malicious_proof
603            .right_boundary_proof
604            .clone_from(&malicious_proof.left_boundary_proof);
605        assert!(tree
606            .verify_namespace_proof(&malicious_proof, 3)
607            .unwrap()
608            .is_err());
609    }
